Можно ли создать функцию, которая будет вести себя по-разному в зависимости от типа аргументов, указанного явно? В некоторой степени это возможно даже в C89. Однако для получения плавного результата нам понадобится довольно сложный механизм макросов.
Во-первых, нужно понимать, как работает этот необычный символ # в контексте макроса. При использовании внутри макроса символ # заключает содержимое символа в кавычки.  

```
#define mystr hello
#define res #mystr
puts( res );  /* will be replaced with `puts("hello")`
```  

Оператор ## представляет ещё больший интерес. Он позволяет динамически формировать имена символов.
```
#define x1 "Hello"
#define x2 " World"
#define str(i) x##i
puts( str(1) );  /* str(1) -> x1 -> "Hello" */
puts( str(2) );  /* str(2) -> x2 -> " World" */
```  

```
#include <stdio.h>
#include <stdbool.h>

#define pair(T) pair_##T                    // pair(T) when called like that: pair(int) will be replaced by the string pair_int;
#define DEFINE_PAIR(T) struct pair(T) {\    // DEFINE_PAIR(int);
    T fst;\
    T snd;\
};\

bool pair_##T##_any(struct pair(T) pair, bool (*predicate)(T)) {\
    return predicate(pair.fst) || predicate(pair.snd); \
}

#define any(T) pair_##T##_any                // Цель макроса сформировать допустимое имя функции в зависимости от типа;
DEFINE_PAIR(int)
bool is_positive( int x ) { return x > 0; }

int main( int argc, char** argv ) {
    struct pair(int) obj;
    obj.fst = 1;
    obj.snd = -1;
    printf("%d\n", any(int)(obj, is_positive) );
    return 0;
}
```

DEFINE_PAIR(int):  
```
struct pair_int {
    int fst;
    int snd;
};
bool pair_int_any(struct pair_int pair, bool (*predicate)(int)) {
    return predicate(pair.fst) || predicate(pair.snd);
}
```
